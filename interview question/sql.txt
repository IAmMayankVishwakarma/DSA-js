4. Find Duplicate Emails
SELECT email, COUNT(*) AS total
FROM employees
GROUP BY email
HAVING COUNT(*) > 1;
5. Delete Duplicate Records
DELETE e1
FROM employees e1
JOIN employees e2
ON e1.email = e2.email
AND e1.id > e2.id;
6. Second Highest Salary
SELECT MAX(salary) AS second_highest
FROM employees
WHERE salary <
(
SELECT MAX(salary)
FROM employees
);
7. Third Highest Salary
SELECT DISTINCT salary
FROM employees
ORDER BY salary DESC
LIMIT 1 OFFSET 2;
8.Highest Salary Department Wise
SELECT department,
MAX(salary)
FROM employees
GROUP BY department;
9.Employees Without Orders
SELECT *
FROM users
LEFT JOIN orders
ON users.id=orders.user_id
WHERE orders.id IS NULL;
10. Total Orders Per User
SELECT users.name,
COUNT(orders.id) total_orders
FROM users
LEFT JOIN orders
ON users.id=orders.user_id
GROUP BY users.id;
11. Last 30 Days Orders
SELECT *
FROM orders
WHERE created_at >= DATE_SUB(NOW(), INTERVAL 30 DAY);
12 Top 5 Highest Paid Employees
SELECT *
FROM employees
ORDER BY salary DESC
LIMIT 5;
13. Find Nth Highest Salary
SELECT DISTINCT salary
FROM employees
ORDER BY salary DESC
LIMIT 1 OFFSET N-1;
