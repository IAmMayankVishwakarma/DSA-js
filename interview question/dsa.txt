method 2

$str = "madam";

$left = 0;
$right = strlen($str) - 1;

$isPalindrome = true;

while ($left < $right) {
    if ($str[$left] != $str[$right]) {
        $isPalindrome = false;
        break;
    }
    $left++;
    $right--;
}

echo $isPalindrome ? "Palindrome" : "Not Palindrome";

=======================================================
sort a array without any built in function 
=====================================================
$array = [5,2,9,1,7];

$n = count($array);

for ($i = 0; $i < $n - 1; $i++) {

    for ($j = 0; $j < $n - $i - 1; $j++) {

        if ($array[$j] > $array[$j + 1]) {

            $temp = $array[$j];
            $array[$j] = $array[$j + 1];
            $array[$j + 1] = $temp;
        }
    }
}

print_r($array);
